const normalizePdfText = (value = '') => String(value)
  .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '')
  .replace(/[\uFFFD]/g, '')
  .replace(/&amp;/gi, '&')
  .replace(/&nbsp;/gi, ' ')
  .replace(/[\u0005\u0010]/g, '')
  .replace(/^[\s'"“”‘’]+|[\s'"“”‘’]+$/g, '')
  .replace(/\s+/g, ' ')
  .trim();

const pdfKnownCriteria = (m = {}) => {
  const map = new Map();
  const add = (id, titulo, evidencia = '', atendeu = null) => {
    const key = normalizePdfText(id || titulo);
    const title = normalizePdfText(titulo || id);
    if (!key || !title) return;
    const previous = map.get(key) || {};
    map.set(key, {
      id: id || previous.id || key,
      titulo: title,
      evidencia: normalizePdfText(evidencia || previous.evidencia || ''),
      atendeu: typeof atendeu === 'boolean' ? atendeu : previous.atendeu
    });
  };

  (Array.isArray(m.criterios) ? m.criterios : []).forEach((c, i) => {
    add(c?.id || `c${i}`, c?.titulo || c?.id, '', null);
  });

  Object.entries(m.detalhes || {}).forEach(([id, r]) => {
    add(id, r?.titulo || id, r?.evidencia || '', r?.atendeu);
  });

  Object.entries(m.respostas || {}).forEach(([id, r]) => {
    const existing = map.get(normalizePdfText(id));
    add(id, existing?.titulo || id, r?.evidencia || '', r?.atendeu);
  });

  return [...map.values()];
};

const splitPdfBullets = (value = '') => {
  const raw = normalizePdfText(value)
    .replace(/^[^A-Za-zÀ-ÿ0-9]+/, '')
    .replace(/[&Ø][^A-Za-zÀ-ÿ]*$/g, '')
    .trim();

  if (!raw) return [];

  return raw
    .split(/\s+-\s+(?=[A-ZÀ-Ú0-9])/)
    .map(x => normalizePdfText(x).replace(/^[-•*]+\s*/, '').trim())
    .filter(Boolean);
};

const extractPdfSummary = (value = '') => {
  const raw = normalizePdfText(value);
  if (!raw) return { positives: [], improvements: [], action: '' };

  const positiveMatch = raw.match(/Pontos\s+Positivos?\s*:(.*?)(?=Oportunidades\s+de\s+Melhoria|Pontos\s+de\s+Aten[cç][aã]o|Plano\s+de\s+A[cç][aã]o|$)/i);
  const improvementMatch = raw.match(/(?:Oportunidades\s+de\s+Melhoria|Pontos\s+de\s+Aten[cç][aã]o)\s*:(.*?)(?=Plano\s+de\s+A[cç][aã]o|$)/i);
  const actionMatch = raw.match(/Plano\s+de\s+A[cç][aã]o\s*:(.*)$/i);

  return {
    positives: splitPdfBullets(positiveMatch?.[1] || ''),
    improvements: splitPdfBullets(improvementMatch?.[1] || ''),
    action: normalizePdfText(actionMatch?.[1] || '')
  };
};

const buildPdfCriteria = (m = {}) => {
  const known = pdfKnownCriteria(m);
  const byTitle = new Map(known.map(r => [normalizePdfText(r.titulo).toLowerCase(), r]));
  const summary = extractPdfSummary(m.feedback || m.resumo || m.resumoCriteriosTexto || '');
  const rows = [];

  const addSummary = (item, atendeu) => {
    let title = normalizePdfText(item);
    let evidence = '';
    const match = title.match(/^(.+?)\s*\((.+)\)$/);
    if (match) {
      title = normalizePdfText(match[1]);
      evidence = normalizePdfText(match[2]);
    }

    const knownRow = byTitle.get(title.toLowerCase()) || known.find(r => {
      const a = normalizePdfText(r.titulo).toLowerCase();
      const b = title.toLowerCase();
      return a === b || a.startsWith(`${b} `) || b.startsWith(`${a} `);
    });

    if (knownRow) {
      const duplicate = rows.find(r => normalizePdfText(r.titulo).toLowerCase() === normalizePdfText(knownRow.titulo).toLowerCase());
      if (duplicate) {
        duplicate.atendeu = atendeu;
        duplicate.evidencia = evidence || duplicate.evidencia || knownRow.evidencia || '';
        return;
      }
      rows.push({ ...knownRow, atendeu, evidencia: evidence || knownRow.evidencia || '' });
      return;
    }

    rows.push({ id: `summary-${rows.length}`, titulo: title, atendeu, evidencia });
  };

  summary.positives.forEach(x => addSummary(x, true));
  summary.improvements.forEach(x => addSummary(x, false));

  known.forEach(r => {
    if (typeof r.atendeu !== 'boolean') return;
    const exists = rows.find(x => normalizePdfText(x.titulo).toLowerCase() === normalizePdfText(r.titulo).toLowerCase());
    if (!exists) rows.push(r);
  });

  return { rows, summary };
};

const makePdf = (m, supervisor) => {
  const J = window.jspdf?.jsPDF || window.jsPDF;
  if (!J) throw new Error('Biblioteca PDF não carregada.');

  const pdf = new J({ orientation: 'portrait', unit: 'mm', format: 'a4', compress: true });
  const W = 210, margin = 18, contentW = W - margin * 2;
  const primary = [37, 99, 235];
  const dark = [23, 32, 42];
  const body = [51, 65, 85];
  const muted = [100, 116, 139];
  const line = [226, 232, 240];
  const success = [22, 101, 52];
  const danger = [185, 28, 28];
  const warning = [180, 83, 9];
  let y = 18;
  let pageNo = 1;

  const footer = () => {
    pdf.setDrawColor(...line);
    pdf.line(margin, 284, W - margin, 284);
    pdf.setFont('helvetica', 'normal');
    pdf.setFontSize(7);
    pdf.setTextColor(...muted);
    pdf.text('Carvalima • Central de Qualidade • Uso interno', margin, 290);
    pdf.text(`Página ${pageNo}`, W - margin, 290, { align: 'right' });
  };

  const newPage = () => {
    footer();
    pdf.addPage();
    pageNo += 1;
    y = 18;
  };

  const ensure = (height = 10) => {
    if (y + height > 276) newPage();
  };

  const drawText = (value, size = 9, bold = false, color = body, maxW = contentW, gap = 4.5, x = margin) => {
    const str = normalizePdfText(value) || '—';
    pdf.setFont('helvetica', bold ? 'bold' : 'normal');
    pdf.setFontSize(size);
    pdf.setTextColor(...color);
    const parts = pdf.splitTextToSize(str, maxW);
    parts.forEach(part => {
      ensure(gap + 1);
      pdf.text(part, x, y);
      y += gap;
    });
  };

  const section = (title) => {
    ensure(13);
    y += 3;
    pdf.setFillColor(...primary);
    pdf.roundedRect(margin, y - 3.5, 2.5, 6.5, 1.2, 1.2, 'F');
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(11);
    pdf.setTextColor(...dark);
    pdf.text(normalizePdfText(title), margin + 6, y + 1.5);
    y += 8;
  };

  // Cabeçalho institucional
  pdf.setFillColor(...primary);
  pdf.rect(0, 0, W, 7, 'F');
  pdf.setFont('helvetica', 'bold');
  pdf.setFontSize(8.5);
  pdf.setTextColor(...primary);
  pdf.text('CARVALIMA • CENTRAL DE QUALIDADE', margin, y + 2);
  y += 12;

  pdf.setFont('helvetica', 'bold');
  pdf.setFontSize(21);
  pdf.setTextColor(...dark);
  pdf.text('Monitoria de atendimento', margin, y);
  y += 6;

  pdf.setFont('helvetica', 'normal');
  pdf.setFontSize(9);
  pdf.setTextColor(...muted);
  const supName = normalizePdfText(supervisor?.nome || 'Supervisor');
  pdf.text(`Documento para acompanhamento de ${supName}`, margin, y);
  y += 10;

  // Identificação
  const infoY = y;
  pdf.setFillColor(248, 250, 252);
  pdf.setDrawColor(...line);
  pdf.roundedRect(margin, infoY, contentW, 35, 4, 4, 'FD');
  pdf.setFont('helvetica', 'bold');
  pdf.setFontSize(7.5);
  pdf.setTextColor(...muted);
  pdf.text('IDENTIFICAÇÃO DA MONITORIA', margin + 6, infoY + 7);

  const leftX = margin + 6;
  const rightX = margin + 102;
  const row1 = infoY + 14;
  const row2 = infoY + 22;
  const row3 = infoY + 30;

  pdf.setFont('helvetica', 'bold');
  pdf.setFontSize(8.5);
  pdf.setTextColor(...dark);
  pdf.text(`ID: ${normalizePdfText(m.id || '—')}`, leftX, row1);
  pdf.text(`Colaborador: ${normalizePdfText(m.agente || '—')}`, rightX, row1);
  pdf.text(`Data: ${normalizePdfText(m.dataAtendimento || '—')}`, leftX, row2);
  pdf.text(`Unidade: ${normalizePdfText(m.unidade || '—')}`, rightX, row2);
  pdf.text(`Departamento: ${normalizePdfText(m.departamento || '—')}`, leftX, row3);
  pdf.text(`Canal: ${normalizePdfText(m.canal || 'Telefone')}`, rightX, row3);
  y = infoY + 43;

  // Resultado
  const note = Number(m.nota || 0).toFixed(1);
  const status = normalizePdfText(m.status || (Number(m.nota || 0) >= 9 ? 'Conforme' : Number(m.nota || 0) >= 7 ? 'Em Atenção' : 'Crítico'));
  pdf.setFillColor(...primary);
  pdf.roundedRect(margin, y, contentW, 27, 4, 4, 'F');
  pdf.setFont('helvetica', 'bold');
  pdf.setFontSize(7.5);
  pdf.setTextColor(219, 234, 254);
  pdf.text('RESULTADO DA MONITORIA', margin + 6, y + 7);
  pdf.setFontSize(17);
  pdf.setTextColor(255, 255, 255);
  pdf.text(`Nota ${note}/10`, margin + 6, y + 19);
  pdf.setFontSize(9);
  pdf.text(status.toUpperCase(), W - margin - 6, y + 18, { align: 'right' });
  y += 35;

  const { rows, summary } = buildPdfCriteria(m);
  const positives = rows.filter(r => r.atendeu === true);
  const improvements = rows.filter(r => r.atendeu === false);

  // Resumo executivo
  section('Resumo da avaliação');
  const total = rows.length;
  const positiveCount = positives.length;
  const improvementCount = improvements.length;
  const summaryText = total
    ? `Foram avaliados ${total} critérios: ${positiveCount} atendido(s) e ${improvementCount} oportunidade(s) de melhoria.`
    : 'A avaliação não possui critérios estruturados disponíveis neste registro.';
  drawText(summaryText, 9, false, body, contentW, 5);

  if (positives.length) {
    ensure(12);
    y += 2;
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(9.5);
    pdf.setTextColor(...success);
    pdf.text('Pontos positivos', margin, y);
    y += 6;
    positives.forEach(r => {
      drawText(`• ${r.titulo}`, 8.7, false, body, contentW - 5, 4.7, margin + 2);
    });
  }

  if (improvements.length) {
    ensure(12);
    y += 2;
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(9.5);
    pdf.setTextColor(...danger);
    pdf.text('Oportunidades de melhoria', margin, y);
    y += 6;
    improvements.forEach(r => {
      drawText(`• ${r.titulo}${r.evidencia ? ` — ${r.evidencia}` : ''}`, 8.7, false, body, contentW - 5, 4.7, margin + 2);
    });
  }

  // Critérios detalhados
  if (rows.length) {
    section('Critérios avaliados');
    rows.forEach((r, i) => {
      const evidence = normalizePdfText(r.evidencia || '');
      const titleLines = pdf.splitTextToSize(`${i + 1}. ${normalizePdfText(r.titulo || r.id)}`, contentW - 47);
      const evidenceLines = evidence ? pdf.splitTextToSize(`Registro: ${evidence}`, contentW - 10) : [];
      const rowH = 13 + Math.min(evidenceLines.length, 2) * 4.2;
      ensure(rowH + 2);
      pdf.setFillColor(i % 2 ? 255 : 248, i % 2 ? 255 : 250, i % 2 ? 255 : 252);
      pdf.setDrawColor(...line);
      pdf.roundedRect(margin, y - 2, contentW, rowH, 2.5, 2.5, 'FD');

      pdf.setFont('helvetica', 'bold');
      pdf.setFontSize(8.2);
      pdf.setTextColor(...dark);
      pdf.text(titleLines.slice(0, 2), margin + 5, y + 5);

      const statusText = r.atendeu === true ? 'ATENDEU' : r.atendeu === false ? 'NÃO ATENDEU' : 'NÃO INFORMADO';
      const statusColor = r.atendeu === true ? success : r.atendeu === false ? danger : warning;
      pdf.setFont('helvetica', 'bold');
      pdf.setFontSize(7.2);
      pdf.setTextColor(...statusColor);
      pdf.text(statusText, W - margin - 5, y + 5, { align: 'right' });

      if (evidenceLines.length) {
        pdf.setFont('helvetica', 'normal');
        pdf.setFontSize(7.2);
        pdf.setTextColor(...muted);
        pdf.text(evidenceLines.slice(0, 2), margin + 5, y + 10.5);
      }

      y += rowH + 2;
    });
  }

  // Plano de ação
  const actionText = normalizePdfText(m.planoAcao || summary.action || '');
  if (actionText) {
    section('Plano de ação');
    const actionLines = pdf.splitTextToSize(actionText, contentW - 12);
    const actionH = Math.max(20, actionLines.length * 4.6 + 15);
    ensure(actionH + 4);
    pdf.setFillColor(248, 250, 252);
    pdf.setDrawColor(...line);
    pdf.roundedRect(margin, y - 2, contentW, actionH, 4, 4, 'FD');
    pdf.setFont('helvetica', 'normal');
    pdf.setFontSize(8.6);
    pdf.setTextColor(...body);
    pdf.text(actionLines, margin + 6, y + 6);
    y += actionH + 3;
    drawText(`Prazo: ${normalizePdfText(m.prazoPlano || 'Não definido')} • Status: ${normalizePdfText(m.statusPlano || 'Pendente')}`, 8, true, muted, contentW, 4.5);
  }

  // Retorno do supervisor
  section('Retorno do supervisor');
  const supervisorDecision = normalizePdfText(m.supervisorDecisao || '');
  const supervisorFeedback = normalizePdfText(m.feedbackSupervisor || '');
  const nextStep = normalizePdfText(m.proximoPassoSupervisor || '');
  const hasReturn = supervisorDecision || supervisorFeedback || nextStep || m.supervisorRetornoAt;

  if (hasReturn) {
    const returnParts = [];
    if (supervisorDecision) returnParts.push(`Decisão: ${supervisorDecision}`);
    if (supervisorFeedback) returnParts.push(`Comentário: ${supervisorFeedback}`);
    if (nextStep) returnParts.push(`Próximo passo: ${nextStep}`);
    if (m.supervisorRetornoAt) returnParts.push(`Respondido em: ${normalizePdfText(new Date(m.supervisorRetornoAt).toLocaleString('pt-BR'))}`);
    const returnLines = returnParts.flatMap(x => pdf.splitTextToSize(x, contentW - 12));
    const returnH = Math.max(24, returnLines.length * 4.5 + 12);
    ensure(returnH + 2);
    pdf.setFillColor(248, 250, 252);
    pdf.setDrawColor(...line);
    pdf.roundedRect(margin, y - 2, contentW, returnH, 4, 4, 'FD');
    pdf.setFont('helvetica', 'normal');
    pdf.setFontSize(8.5);
    pdf.setTextColor(...body);
    pdf.text(returnLines, margin + 6, y + 7);
    y += returnH + 4;
  } else {
    ensure(27);
    pdf.setFillColor(248, 250, 252);
    pdf.setDrawColor(...line);
    pdf.roundedRect(margin, y - 2, contentW, 23, 4, 4, 'FD');
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(8.5);
    pdf.setTextColor(...warning);
    pdf.text('AGUARDANDO RETORNO', margin + 6, y + 7);
    pdf.setFont('helvetica', 'normal');
    pdf.setFontSize(8.2);
    pdf.setTextColor(...body);
    const waiting = pdf.splitTextToSize('O supervisor deverá registrar ciência, orientação e próximos passos pelo link enviado por e-mail.', contentW - 12);
    pdf.text(waiting, margin + 6, y + 14);
    y += 30;
  }

  footer();
  return pdf.output('datauristring').split(',')[1];
};
